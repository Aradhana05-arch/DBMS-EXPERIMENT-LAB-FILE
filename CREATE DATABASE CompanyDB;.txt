CREATE DATABASE CompanyDB;
USE CompanyDB;

CREATE TABLE Department (
    dept_id INT PRIMARY KEY,
    dept_name VARCHAR(50) NOT NULL,
    location VARCHAR(50)
);

CREATE TABLE Project (
    project_id INT PRIMARY KEY,
    project_name VARCHAR(100) NOT NULL,
    budget DECIMAL(12,2)
);

CREATE TABLE Employee (
    emp_id INT PRIMARY KEY,
    emp_name VARCHAR(50) NOT NULL,
    salary DECIMAL(10,2),
    dept_id INT,
    project_id INT,
    joining_date DATE,
    FOREIGN KEY (dept_id) REFERENCES Department(dept_id),
    FOREIGN KEY (project_id) REFERENCES Project(project_id)
);

    AVG(salary) AS average_salary,

    MIN(salary) AS lowest_salary

SELECT
    dept_id,
    COUNT(emp_id) AS employee_count
FROM Employee

    d.dept_name,
    AVG(e.salary) AS average_salary
FROM Employee e
INNER JOIN Department d
ON e.dept_id = d.dept_id

ON e.dept_id = d.dept_id

CREATE VIEW Dept_Salary_Report AS
SELECT
    d.dept_id,
    d.dept_name,
    COUNT(e.emp_id) AS employee_count,
    SUM(e.salary) AS salary_total,
    AVG(e.salary) AS salary_average
FROM Department6 d
LEFT JOIN Employee6 e
ON d.dept_id = e.dept_id
GROUP BY d.dept_id, d.dept_name;

WITH RECURSIVE StaffChain AS (
    SELECT
        emp_id,
        emp_name,
        manager_id,
        0 AS level,
        CAST(emp_name AS CHAR(200)) AS reporting_path
    FROM Employee6
    WHERE manager_id IS NULL

    UNION ALL

    SELECT
        e.emp_id,
        e.emp_name,
        e.manager_id,
        sc.level + 1,
        CONCAT(sc.reporting_path, ' -> ', e.emp_name)
    FROM Employee6 e
    INNER JOIN StaffChain sc
    ON e.manager_id = sc.emp_id
)

SELECT
    emp_id,
    emp_name,
    manager_id,
    level,
    reporting_path
FROM StaffChain
ORDER BY reporting_path;

WITH RECURSIVE StaffChain AS (

    FROM Employee6

        e.emp_name,
        e.manager_id,
        sc.level + 1
    FROM Employee6 e
    JOIN StaffChain sc
    ON e.manager_id = sc.emp_id

SELECT

    emp_name,
    manager_id,
    level
FROM StaffChain

CREATE TRIGGER department_change_audit7
AFTER UPDATE ON Employee7
FOR EACH ROW
BEGIN
    IF OLD.dept_id <> NEW.dept_id THEN
        INSERT INTO Employee_Audit7
        (emp_id, previous_dept, updated_dept)
        VALUES
        (OLD.emp_id, OLD.dept_id, NEW.dept_id);
    END IF;
END //

DELIMITER ;

CREATE PROCEDURE change_employee_department(
    IN p_employee_id INT,
    IN p_department_id INT
)
BEGIN
    DECLARE v_count INT DEFAULT 0;

    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Error occurred during department transfer';
    END;

    START TRANSACTION;

    SELECT COUNT(*)
    INTO v_count
    FROM Employee7
    WHERE emp_id = p_employee_id;

    IF v_count = 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Employee does not exist';
    END IF;

    SELECT COUNT(*)
    INTO v_count
    FROM Department7
    WHERE dept_id = p_department_id;

    IF v_count = 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Department does not exist';
    END IF;

    UPDATE Employee7
    SET dept_id = p_department_id
    WHERE emp_id = p_employee_id;

    COMMIT;
END //

DELIMITER ;
